// Helper chuyển X-API (PlatformData / DataSet) → JSON, dùng NGAY TRONG backend Nexacro thật của dự án
// (nơi đã có thư viện X-API). Cách B trong docs/nexacro-migration/NEXACRO_XAPI_TO_JSON.md.
//
// File .txt để không bị biên dịch trong repo này (không có thư viện X-API).
// Tên lớp / hàm đã đối chiếu với Javadoc X-API Nexacro 17 (package com.nexacro17.xapi.data):
// http://docs.tobesoft.com/xapi_java_nexacro_17_ko/index-all.html
// Bản khác (Nexacro 14 / N) có thể khác tên package: đối chiếu Javadoc X-API của dự án.

import com.nexacro17.xapi.data.ColumnHeader;
import com.nexacro17.xapi.data.DataSet;
import com.nexacro17.xapi.data.DataSetList;
import com.nexacro17.xapi.data.PlatformData;
import com.nexacro17.xapi.data.Variable;
import com.nexacro17.xapi.data.VariableList;

import java.util.*;

public final class XapiJsonConverter {

    private XapiJsonConverter() {
    }

    /**
     * PlatformData → Map (Spring/Jackson tự ghi ra JSON):
     * { "errorCode": 0, "errorMsg": "SUCC", "params": {...}, "datasets": { "ds_list": [ {...}, ... ] } }
     * Cùng định dạng với cổng /api/nx/** trong repo này, để app Android dùng chung NxResponse.
     */
    public static Map<String, Object> toJson(PlatformData data) {
        Map<String, Object> params = new LinkedHashMap<>();
        VariableList vars = data.getVariableList();
        for (int i = 0; i < vars.size(); i++) {
            Variable v = vars.get(i);
            params.put(v.getName(), v.getObject());
        }
        Object code = params.remove("ErrorCode");
        Object msg = params.remove("ErrorMsg");

        Map<String, Object> datasets = new LinkedHashMap<>();
        DataSetList list = data.getDataSetList();
        for (int i = 0; i < list.size(); i++) {
            DataSet ds = list.get(i);
            datasets.put(ds.getName(), rows(ds));
        }

        Map<String, Object> json = new LinkedHashMap<>();
        json.put("errorCode", code == null ? 0 : ((Number) code).intValue());
        json.put("errorMsg", msg == null ? "" : msg.toString());
        json.put("params", params);
        json.put("datasets", datasets);
        return json;
    }

    /** DataSet → danh sách dòng; mỗi dòng Map tên cột → giá trị (kiểu Java X-API đã chuyển sẵn). */
    public static List<Map<String, Object>> rows(DataSet ds) {
        List<Map<String, Object>> rows = new ArrayList<>();
        for (int r = 0; r < ds.getRowCount(); r++) {
            Map<String, Object> row = new LinkedHashMap<>();
            for (int c = 0; c < ds.getColumnCount(); c++) {
                ColumnHeader col = ds.getColumn(c);
                row.put(col.getName(), ds.getObject(r, c));
            }
            int type = ds.getRowType(r);
            if (type == DataSet.ROW_TYPE_INSERTED) row.put("_rowType", "insert");
            if (type == DataSet.ROW_TYPE_UPDATED) row.put("_rowType", "update");
            rows.add(row);
        }
        // Dòng đã xoá nằm riêng (removed rows) trong X-API
        for (int r = 0; r < ds.getRemovedRowCount(); r++) {
            Map<String, Object> row = new LinkedHashMap<>();
            for (int c = 0; c < ds.getColumnCount(); c++) {
                row.put(ds.getColumn(c).getName(), ds.getRemovedData(r, c));
            }
            row.put("_rowType", "delete");
            rows.add(row);
        }
        return rows;
    }
}
